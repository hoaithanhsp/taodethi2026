export const extractDocxTablesAndHtml = async (arrayBuffer: ArrayBuffer): Promise<{
  fullHtml: string;
  matrixHtml: string;
  specsHtml: string | null;

}> {
  const zip = await JSZip.loadAsync(arrayBuffer);
  const docXmlFile = zip.file('word/document.xml');
  
  if (docXmlFile) {

  // 1. Detect subject (KNOWN_SUBJECTS regex list)

      break;

  // 2. Detect grade
  const gradeMatch = plainText.match(/\b(?:LỚP|KHỐI)\s*[:\-]?\s*(1[0-2]|[1-9])\b/i);
  if (gradeMatch) result.grade = gradeMatch[1];
  
  // 3. Parse DOMParser HTML table → 2D grid (expanding rowSpan/colSpan)
  const doc = new DOMParser().parseFromString(htmlOrText, 'text/html');
  const table = doc.querySelector('table');
  
  // Build 2D grid
  const grid: GridCell[][] = [];
  for (let r = 0; r < trList.length; r++) {
    for (const cell of trList[r].cells) {
      const rs = cell.rowSpan || 1;

        for (let dc = 0; dc < cs; dc++) {
          grid[r + dr][cCursor + dc] = { text: cellText };
        }
      }
    }
  }
  
  // 4. Identify header rows (rows 0-4 with keywords)
  let headerRows = 1;
  for (let r = 0; r < Math.min(5, numRows); r++) {
    if (/\b(tt|chương|mức độ|biết|hiểu|vận dụng)\b/i.test(grid[r].join(' '))) {
      headerRows = r + 1;
    }
  }
  
  // 5. Map columns: question type (type1/type2/type3/essay) + level (biet/hieu/van_dung/van_dung_cao)
  const colMapping = new Map<number, { type: QTypeKey; level: LevelKey }>();
  for (let c = 0; c < numCols; c++) {
    const fullHeader = grid.slice(0, headerRows).map(row => row[c]?.text).join(' | ');
    
    let qType: QTypeKey | null = null;
    if (/nhiều\s*lựa\s*chọn|phần\s*i\b|dạng\s*i\b/i.test(fullHeader)) qType = 'type1';
    else if (/đúng\s*sai|phần\s*ii\b|dạng\s*ii\b/i.test(fullHeader)) qType = 'type2';
    else if (/trả\s*lời\s*ngắn|phần\s*iii\b|dạng\s*iii\b/i.test(fullHeader)) qType = 'type3';
    else if (/tự\s*luận|phần\s*iv\b|dạng\s*iv\b/i.test(fullHeader)) qType = 'essay';
    
    let level: LevelKey | null = null;

    else if (/vận\s*dụng|\bvd\b/i.test(fullHeader)) level = 'van_dung';
    else if (/thông\s*hiểu|\bth\b/i.test(fullHeader)) level = 'hieu';
    else if (/nhận\s*biết|\bnb\b/i.test(fullHeader)) level = 'biet';
    
    if (qType && level) colMapping.set(c, { type: qType, level });
  }
  
  // 6. Count questions per cell (regex: "I.1", "Câu 1", integer numbers)
  for (let r = headerRows; r < numRows; r++) {

    }
  }
  
  return { subject, grade, questionConfig, examType, duration };
